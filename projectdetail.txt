alur: truk datang, ada trigger untuk memotret truk (yang ini bukan urusanmu) -> gambar masuk folder input -> langsung cari plat (method proses cari plat + plat.pt) -> plat nemu, crop lokal, lakukan preprocessing (method preprocessing) -> langsung cari karakter terbaca (method proses cari karakter + char.pt) -> data hasil bacaan disimpan ke JSON -> create gambar yang ada label nya simpan ke folder output

DETAIL JSON
json berisi data setiap kendaraan berdasarkan input gambar. jadi gambar masuk itu dihitung sebagai sebuah kendaraan yang baru aja masuk. catat timestamp dan plat nomor terbaca. kemudian catat juga path gambar input dan path gambar outputnya. jangan lupa id nya. jadi antara id itu penamaannya sama dengan foto pada input dan pada output. boleh cuma angka 1, 2, 3 gitu doang apapa.

METHOD PREPROCESSING
gunakan code dibawah ini sebagai acuan algoritma preprocessing (jangan  ganti2 algoritma nya)
import cv2
import matplotlib.pyplot as plt
import numpy as np

class NLMDenoiseProcessor:
    def __init__(self):
        self.clahe = cv2.createCLAHE(clipLimit=4.0, tileGridSize=(4,4))
    
    def load_image(self, image_path):
        """Load image dan convert ke RGB"""
        image = cv2.imread(image_path)
        return cv2.cvtColor(image, cv2.COLOR_BGR2RGB)
    
    def average_grayscale(self, image):
        """Convert ke grayscale menggunakan average method"""
        blue = image[:,:,0]
        green = image[:,:,1] 
        red = image[:,:,2]
        return blue/3 + green/3 + red/3
    
    def apply_clahe(self, grayscale_image):
        """Apply CLAHE contrast enhancement"""
        grayscale_uint8 = np.uint8(grayscale_image)
        return self.clahe.apply(grayscale_uint8)
    
    def gamma_correction(self, image, gamma=1.3):
        """Apply gamma correction"""
        inv_gamma = 1.0 / gamma
        table = np.array([((i / 255.0) ** inv_gamma) * 255 for i in range(256)]).astype("uint8")
        return cv2.LUT(image, table)
    
    def nlm_denoise(self, image, h=15, template_window=7, search_window=21):
        """Apply Non-Local Means Denoising - Fokus penelitian"""
        return cv2.fastNlMeansDenoising(image, None, h, template_window, search_window)

# ============================================================================
# MAIN PROCESSING PIPELINE
# ============================================================================
def main():
    processor = NLMDenoiseProcessor()
    
    # Load image
    image_path = r"D:\Work\PT TPS\ALPR Model Training\ALPR_MODEL_PREPROCESSING\output\plat_15.jpg"
    original_image = processor.load_image(image_path)
    
    # Processing pipeline
    grayscale = processor.average_grayscale(original_image)
    clahe_enhanced = processor.apply_clahe(grayscale)
    gamma_corrected = processor.gamma_correction(clahe_enhanced)
    nlm_denoised = processor.nlm_denoise(gamma_corrected)
    
    # Convert to uint8 for display
    grayscale_uint8 = np.uint8(grayscale)
    
    # Visualization
    plt.figure(figsize=(15, 10))
    
    # Original Image
    plt.subplot(2, 3, 1)
    plt.imshow(original_image)
    plt.title('1. Original Image')
    plt.axis('off')
    
    # Grayscale
    plt.subplot(2, 3, 2)
    plt.imshow(grayscale, cmap='gray')
    plt.title('2. Grayscale Conversion')
    plt.axis('off')
    
    # CLAHE Enhanced
    plt.subplot(2, 3, 3)
    plt.imshow(clahe_enhanced, cmap='gray')
    plt.title('3. CLAHE Contrast Enhancement')
    plt.axis('off')
    
    # Gamma Correction
    plt.subplot(2, 3, 4)
    plt.imshow(gamma_corrected, cmap='gray')
    plt.title('4. Gamma Correction\n(γ=1.3)')
    plt.axis('off')
    
    # NLM Denoise - FOKUS SKRIPSI
    plt.subplot(2, 3, 5)
    plt.imshow(nlm_denoised, cmap='gray')
    plt.title('5. NLM Denoising\n(Fokus Penelitian)')
    plt.axis('off')
    
    # Final Comparison
    plt.subplot(2, 3, 6)
    comparison = np.hstack((grayscale_uint8, nlm_denoised))
    plt.imshow(comparison, cmap='gray')
    plt.title('6. Comparison\n(Left: Original, Right: Processed)')
    plt.axis('off')
    
    plt.tight_layout()
    plt.show()
    
    # ============================================================================
    # STATISTICAL ANALYSIS - FOKUS NLM DENOISING
    # ============================================================================
    print("=== PROCESSING STATISTICS ===")
    print(f"{'1. Grayscale':<25} - Min: {grayscale_uint8.min():3d}, Max: {grayscale_uint8.max():3d}, Mean: {grayscale_uint8.mean():6.2f}")
    print(f"{'2. CLAHE Enhanced':<25} - Min: {clahe_enhanced.min():3d}, Max: {clahe_enhanced.max():3d}, Mean: {clahe_enhanced.mean():6.2f}")
    print(f"{'3. Gamma Corrected':<25} - Min: {gamma_corrected.min():3d}, Max: {gamma_corrected.max():3d}, Mean: {gamma_corrected.mean():6.2f}")
    print(f"{'4. NLM Denoised':<25} - Min: {nlm_denoised.min():3d}, Max: {nlm_denoised.max():3d}, Mean: {nlm_denoised.mean():6.2f}")
    
    # Noise reduction analysis
    noise_reduction = abs(gamma_corrected.std() - nlm_denoised.std())
    print(f"{'5. Noise Reduction':<25} - {noise_reduction:.2f} (std dev reduction)")
    
    # PSNR calculation (jika diperlukan)
    mse = np.mean((gamma_corrected - nlm_denoised) ** 2)
    if mse > 0:
        psnr = 20 * np.log10(255.0 / np.sqrt(mse))
        print(f"{'6. PSNR':<25} - {psnr:.2f} dB")

if __name__ == "__main__":
    main()
